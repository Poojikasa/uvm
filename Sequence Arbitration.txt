UVM Sequence Arbitration — Question
Write only the body() implementations for the following four sequences. Do not write the test or driver code.
1. xseq_high — Set priority to 500. Generate 5 sequence items. Use m_sequencer.lock() and m_sequencer.unlock() to ensure that the first 3 items execute without interruption.
2. xseq_medium — Set priority to 300. Generate 6 sequence items and implement arbitration after every 9 items.
3. xseq_low — Set priority to 200. Generate 4 sequence items, but wait until at least one item from xseq_high has executed before starting.
4. xseq_burst — Set priority to 400. Generate 3 sequence items using m_sequencer.grab() and m_sequencer.ungrab() to obtain exclusive access.
For every sequence item, display the sequence name, transaction number, and simulation time using $display().

class rseq_high extends uvm_sequence#(txn);
`uvm_object_utils(rseq_high)
function new(string name="rseq_high");
super.new(name);
endfunction
task body();
my_sequencer sqr;
txn t;
m_sequencer.lock(this);
for(int i=1;i<=3;i++)
begin
t=txn::type_id::create("t");
start_item(t);
$display("rseq_high txn=%0d time=%0t",i,$time);
if(i==1)
begin
$cast(sqr,m_sequencer);
sqr.high_started=1;
end
finish_item(t);
end
m_sequencer.unlock(this);


for(int i=1;i<=2;i++)
begin
txn t;
t=txn::type_id::create("t");
m_sequencer.lock(this);
start_item(t);
$display("rseq high txn=%0d time=%0t",i,$time);
m_sequencer.unlock(this);
finish_item(t);
end
endtask
endclass

rseq_medium s_medium;
rseq_low s_low;
rseq_burst s_burst;

s_burst=rseq_burst::type_id::create("s_burst");

fork
$display("arbitration=%p",env.sqr.get_arbitration());
s_high.start(env.sqr,,500);
s_medium.start(env.sqr,,300);
s_low.start(env.sqr,,200);
s_burst.start(env.sqr,,400);
join
phase.drop_objection(this);
//end
endtask
endclass
