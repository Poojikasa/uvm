class base_driver extends uvm_driver;
`uvm_component_utils(base_driver)
function new(string name="base_driver",uvm_component parent);
super.new(name,parent);
endfunction
task run_phase(uvm_phase phase);
`uvm_info("BASE_DRV,base driver is running,UVM_LOW)
endtask
endclass 

class new_driver extends base_driver;
`uvm_component_utils(new_driver)
function new(string name="new_driver",uvm_component parent);
super.new(name,parent);
endfunction
task run_phase(uvm_phase phase);
`uvm_info("NEW_DRV,base driver is running,UVM_LOW)
endtask
endclass 


class agent extends uvm_agent;
`uvm_component_utils(agent)
base_driver drv;
function new(string name="agent",uvm_component parent);
super.new(name,parent);
endfunction
function void build_phase(uvm_phase phase);
super.build_phase(phase);
drv=base_driver::type_id::create("drv",this);
endfunction
endclass

class env extends uvm_env;
`uvm_component_utils(env)
agent agt[];
function new(string name="env",uvm_component parent);
super.new(name,parent);
endfunction
function void build_phase(uvm_phase phase);
super.build_phase(phase);
foreach(agt[i])
begin
agt[i]=agent::type_id::create($sformatf(["%0d"],i),this);
end
endfunction
endclass

class test extends uvm_test;
`uvm_component_utils(test)
env eh;
function new(string name="test",uvm_component parent);
super.new(name,parent);
endfunction
function void build_phase(uvm_phase phase);
super.build_phase(phase);
set_type_override_by_type("eh.agt[1].drv",base_driver::get_type(),new_driver::get_type());
endfunction
endclass
